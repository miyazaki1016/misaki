## 目的
本番と演技研究室が同じ Relationship Interpreter / Gemini generation boundary を共有できる第一段階を作る。

## 現在の実装
- 5軸＋親密度＋明示的 relationship status から、台詞ではなく演技指示を生成する shared Relationship Interpreter を追加
- A〜E のオーディション profile と no-write 演技研究室 POST endpoint を追加
- 演技研究室は Production では 404。Preview の公開 GET trigger は削除済み
- 本番 chat と演技研究室の Gemini 呼び出しを shared generator へ統合
- shared generator は既存本番と同じ 30s timeout、timeout 1回 retry、429/502/503/504 の 2s/5s retry を維持
- 本番 chat から shared Relationship Interpreter を呼ぶ経路を追加
- 現在の canonical DB は旧 relationship_points のみなので、legacy adapter は親密度だけを変換し、友情・信頼・じゃれ合い・親愛・恋愛・交際状態を推測しない
- 旧 relationship guide は当面併設。ただし points だけでは決められない嫉妬・甘え・好意・照れ・じゃれ合い表現を除去し、距離感担当へ整理
- safety contract / legacy adapter tests を追加
- 既存 source-string retry tests は shared generator を source of truth として追従

## 安全契約
- score から台詞を直結しない
- romance score だけで交際成立させない
- 時間経過だけで恋しさ・寂しさ・嫉妬・恋愛感情を作らない
- legacy relationship_points から5軸を捏造しない
- 演技研究室は quota/history/memory/relationship/Body Clock を書き込まない

- Canonical relationship integration tests #307: SUCCESS
- Vercel Preview: READY
- 演技研究室 A〜E: HTTP 200 / writes:false / 全profile生成成功
- PR mergeable: true

## このPRでまだやらないこと
- 5軸の canonical DB schema / 保存・更新ロジック
- 6段階親密度の正式閾値確定とハートUI
- explicit relationship status の canonical persistence

+const MODEL = "gemini-3.1-flash-lite";
+
+export async function generateGeminiJson(
+  request: GeminiJsonRequest
+): Promise<GeminiJsonResponse> {
+  const body = JSON.stringify({
+    systemInstruction: {
+      parts: [{ text: request.systemInstruction }],
+    },
+    contents: [
+      ...request.contents,
+      {
+        role: "user",
+        parts: [{ text: request.userText }],
+      },
+    ],
+    generationConfig: {
+      responseMimeType: "application/json",
+    },
+  });
+

+    request.transientRetryDelaysMs ?? [2_000, 5_000];
+  const timeoutRetryDelaysMs =
+    request.timeoutRetryDelaysMs ?? [2_000];
+  let transientAttempt = 0;
+  let timeoutAttempt = 0;
+
+  while (true) {
+    const controller = new AbortController();
+    const timeout = setTimeout(
+      () => controller.abort(),
+      request.timeoutMs ?? 30_000
+    );
+
+    let response: Response;
+    try {
+      response = await fetch(
+        `https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent?key=${request.apiKey}`,
+        {
+          method: "POST",
+          headers: { "Content-Type": "application/json" },
+          signal: controller.signal,
+          body,
+        }
+      );
+    } catch (error) {
+      if (
+        error instanceof Error &&
+        error.name === "AbortError" &&
+        timeoutAttempt < timeoutRetryDelaysMs.length
+      ) {
+        const delayMs = timeoutRetryDelaysMs[timeoutAttempt++];
+        await new Promise((resolve) => setTimeout(resolve, delayMs));
+        continue;
+      }
+      throw error;
+    } finally {
+      clearTimeout(timeout);
+    }

+      continue;
+    }
+
+    const data = await response.json();
