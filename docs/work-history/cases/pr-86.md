+export function compareLabReviews(baseline: LabReview, candidate: LabReview): LabComparison {
+  if (baseline.scenarioId !== candidate.scenarioId) throw Error("scenario mismatch");
+  for (const review of [baseline, candidate]) {
+    if (!review.reviewer.trim()) throw Error("reviewer required");
+    for (const axis of QUALITY_AXES) {
+      const score = review.scores[axis];
+      if (!Number.isInteger(score) || score < 1 || score > 5) throw Error("invalid review score: " + axis);
+    }
+  }
+  const deltas = {} as Record<LabQualityAxis, number>;
+  const regressions: LabQualityAxis[] = [];
+  for (const axis of QUALITY_AXES) {
+    deltas[axis] = candidate.scores[axis] - baseline.scores[axis];
+    if (deltas[axis] < 0) regressions.push(axis);
+  }
+  return { scenarioId: baseline.scenarioId, baseline, candidate, deltas, regressions };
+}

+    const key = finding.category + ":" + finding.criterionIndex;

+        finding.evidenceTurn < 1 || finding.evidenceTurn > run.completedPairs)) ||
+        !finding.rationale.trim()) {
+      throw new Error("QUALITY_ASSESSMENT_FINDING_INVALID");
+    }
+    seen.add(key);
+  }
+  const outcome = findings.some(f => f.verdict === "fail")
+    ? "fail"
+    : findings.some(f => f.verdict === "uncertain") ? "needs-review" : "pass";
+  return { schemaVersion: 1, probeId: probe.id, runLabel, provenance, reviewer, findings: [...findings], outcome };

+
+export function compareQualityAssessments(
+  baseline: QualityAssessment,
+  candidate: QualityAssessment,
+): { probeId: string; baseline: QualityAssessment["outcome"]; candidate: QualityAssessment["outcome"]; reviewRequired: boolean } {
+  if (baseline.probeId !== candidate.probeId ||
+      baseline.provenance !== "recorded-model-output" ||
+      candidate.provenance !== "recorded-model-output" ||
+      baseline.runLabel === candidate.runLabel) {
+    throw new Error("QUALITY_COMPARISON_INCOMPATIBLE");

+  // Human judgement is not reduced to fabricated improvement percentages.

+    probeId: baseline.probeId,

+    candidate: candidate.outcome,

+  transcript: readonly LabTurn[];
+  turnIndex: number;
+}) => Promise<string>;
+export type LabRun = {
+  scenarioId: string;
+  turns: LabTurn[];
+  completedPairs: number;
+  status: "complete" | "stopped" | "error";
+  error?: string;
+};
+export type LabSink = (record: LabRun) => Promise<void> | void;
+
+export async function runLabScenario(input: {
+  scenario: LabScenario;
+  reply: LabReply;
+  nextUser: LabUser;
+  pairs?: number;
+  sink?: LabSink;
+}): Promise<LabRun> {
+  const { scenario, reply, nextUser, sink } = input;
+  const pairs = input.pairs ?? scenario.maxTurns;
+  if (!Number.isInteger(pairs) || pairs < 1 || pairs > scenario.maxTurns) {
+    throw new Error("pairs exceed scenario budget");
+  }
+  const turns: LabTurn[] = [];
+  const result: LabRun = { scenarioId: scenario.id, turns, completedPairs: 0, status: "stopped" };
+  try {
+    for (let index = 0; index < pairs; index++) {
+      const message = index === 0
+        ? scenario.opening
+        : await nextUser({ scenario, transcript: turns.map(turn => ({ ...turn })), turnIndex: index });
+      if (typeof message !== "string" || !message.trim() || message.length > 4000) {
+        throw new Error("invalid user turn");
+      }
+      const history = turns.length ? toGeminiHistory(turns) : [];
+      turns.push({ role: "user", text: message });
+      const answer = await reply({ scenario, message, history, turnIndex: index });
+      if (typeof answer !== "string" || !answer.trim() || answer.length > 4000) {
+        throw new Error("invalid model reply");
+      }
+      turns.push({ role: "model", text: answer });
+      validateLabTranscript(turns, scenario.maxTurns);
+      result.completedPairs++;
+    }
+    result.status = "complete";
+  } catch (error) {
+    result.status = "error";
+    result.error = error instanceof Error ? error.message : "unknown error";
+  }
+  if (sink) await sink({ ...result, turns: turns.map(turn => ({ ...turn })) });
+  return result;
+}
+
+export async function runOfflineBaseline(input: {
+  reply: LabReply;
+  nextUser: LabUser;
+  sink?: LabSink;
+  scenarios?: readonly LabScenario[];
+}): Promise<LabRun[]> {
+  const results: LabRun[] = [];
+  for (const scenario of input.scenarios ?? BASELINE_SCENARIOS) {
+    results.push(await runLabScenario({ ...input, scenario }));
+  }
+  return results;
+}
